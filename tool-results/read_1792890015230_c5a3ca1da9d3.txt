   119→/** Entri panel dropdown dengan stagger halus. */
   120→function PanelItem({
   121→  index,
   122→  reduce,
   123→  className,
   124→  children,
   125→}: {
   126→  index: number;
   127→  reduce: boolean;
   128→  className?: string;
   129→  children: React.ReactNode;
   130→}) {
   131→  return (
   132→    <motion.div
   133→      initial={reduce ? false : { opacity: 0, y: 6 }}
   134→      animate={{ opacity: 1, y: 0 }}
   135→      transition={{ delay: 0.028 * index, duration: 0.16, ease: "easeOut" }}
   136→      className={className}
   137→    >
   138→      {children}
   139→    </motion.div>
   140→  );
   141→}
